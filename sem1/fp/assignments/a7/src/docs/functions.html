 83def get_complex_number(number: str) -> list:
 84    """
 85    Parse a string as a complex number and return [real, imag].
 86    :param number: Complex number string, e.g., "3+4i" or "-2-5i"
 87    :return: Complex number [real, imag]
 88    :rtype: list
 89    :raises ValueError: If the string is not a complex number.
 90    """
 91    number = number.replace(" ", "")
 92
 93    # Must end in 'i'
 94    if not number.endswith("i"):
 95        raise ValueError("Number must end with 'i'.")
 96
 97    number = number[:-1]  # remove trailing 'i'
 98
 99    # split into real + imaginary parts
100    parts = re.split(r'(?<!^)(?=[+-])', number)
101
102    if len(parts) != 2:
103        raise ValueError("Invalid complex number format! Expected a+bi or a-bi.")
104
105    try:
106        real = int(parts[0])
107        if parts[1] not in ("+", "-"):
108            imag = int(parts[1])
109        else:
110            if parts[1] == "+":
111                imag = 1
112            else:
113                imag = -1
114    except ValueError:
115        raise ValueError("Real and imaginary part of complex number must be integers!")
116
117    return create_complex_number(real, imag)

236def list_modulo(number_list: list, symbol: str, limit: int) -> list:
237    """
238    Filter numbers by modulus with a comparison operator.
239    :param number_list: List to filter.
240    :param symbol: '<', '=', or '>'
241    :param limit: Threshold value.
242    :raises ValueError: if 'symbol' is invalid.
243    :return: The list of numbers that satisfy the condition.
244    :rtype: list
245    """
246    if symbol not in ('<', '=', '>'):
247        raise ValueError("Invalid symbol!")
248    list_modulos = []
249    for i in range(len(number_list)):
250        if symbol == '=':
251            if get_modulo(number_list[i]) == limit:
252                list_modulos.append(number_list[i])
253        elif symbol == '<':
254            if get_modulo(number_list[i]) < limit:
255                list_modulos.append(number_list[i])
256        else:
257            if get_modulo(number_list[i]) > limit:
258                list_modulos.append(number_list[i])
259    return list_modulos

276def filter_modulo(number_list: list, symbol: str, limit: int) -> None:
277    """
278    Remove numbers that do not satisfy modulus condition.
279    :param number_list: List to modify.
280    :param symbol: '<', '=', or '>'
281    :param limit: Threshold value.
282    :raises ValueError: if 'symbol' is invalid.
283    :rtype: None
284    """
285    if symbol not in ('<', '=', '>'):
286        raise ValueError("Invalid symbol!")
287
288    i = 0
289    while i < len(number_list):
290        mod = get_modulo(number_list[i])
291        remove = False
292        if symbol == "<":
293            if mod >= limit:  # remove numbers that do not satisfy the condition
294                remove = True
295        elif symbol == "=":
296            if mod != limit:
297                remove = True
298        elif symbol == ">":
299            if mod <= limit:
300                remove = True
301
302        if remove:
303            for j in range(i, len(number_list) - 1):
304                number_list[j] = number_list[j + 1]
305            number_list.pop()
306        else:
307            i += 1

320def undo_state(number_list: list, history: list) -> None:
321    """
322    Undo the last operation by restoring the previous list state.

335def build_numbers_table(numbers: list) -> Texttable:
336    """
337    Build a Texttable representing the list of complex numbers.
338    :param numbers: List of numbers.
339    :return: Textable representing the list of complex numbers.
340    :rtype: Texttable
341    """
342    table = Texttable()
343    table.header(["Index", "Real", "Imag"])
344    table.set_deco(Texttable.HEADER | Texttable.VLINES | Texttable.HLINES)
345
346    for i, c in enumerate(numbers):
347        table.add_row([i, get_real(c), get_imag(c)])
348
